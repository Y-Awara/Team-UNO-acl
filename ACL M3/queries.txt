// 1. Query 1
MATCH (c:Customer)-[:PLACED]->(o:Order)
WHERE o.delivery_delay_days IS NOT NULL
  AND o.delivery_delay_days <= 0
WITH DISTINCT c
RETURN count(c) AS customers_with_at_least_one_ontime_delivery;


// 2. Query 2
MATCH (st:State {name: 'SP'})<-[:LOCATED_IN]-(c:Customer)
MATCH (c)-[:WROTE]->(r:Review)-[:REVIEWS]->(o:Order)
MATCH (o)-[:CONTAINS]->(oi:OrderItem)-[:REFERS_TO]->(p:Product)
WHERE r.review_score IS NOT NULL
  AND p.product_category_name IS NOT NULL
WITH p.product_category_name AS category,
     avg(r.review_score) AS avg_review_score
RETURN category, avg_review_score
ORDER BY avg_review_score DESC
LIMIT 3;


// 3. Query 3
MATCH (st:State)
OPTIONAL MATCH (st)<-[:LOCATED_IN]-(c:Customer)-[:PLACED]->(o:Order)
WITH st, count(DISTINCT o) AS num_orders
RETURN st.name AS state, num_orders
ORDER BY state ASC;


// 4. Query 4
MATCH (s:Seller)<-[:SOLD_BY]-(oi:OrderItem)<-[:CONTAINS]-(o:Order)
WHERE o.delivery_delay_days IS NOT NULL
WITH DISTINCT s, o
WITH s, avg(o.delivery_delay_days) AS avg_delay
RETURN s.seller_id AS seller_id, avg_delay
ORDER BY avg_delay ASC
LIMIT 3;


// 5. Query 5
MATCH (p:Product)<-[:REFERS_TO]-(:OrderItem)<-[:CONTAINS]-(o:Order)<-[:REVIEWS]-(r:Review)
WHERE r.review_score IS NOT NULL
WITH p, avg(r.review_score) AS avg_score
WITH collect({product: p, avg_score: avg_score}) AS rows,
     max(avg_score) AS max_avg
UNWIND rows AS row
WITH row.product AS p, row.avg_score AS avg_score, max_avg
WHERE avg_score = max_avg
RETURN p.product_id AS product_id,
       p.product_category_name AS category,
       avg_score AS average_review_score;
